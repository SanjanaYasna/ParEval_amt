    ```cpp
    /* Forward FFT using iterative Cooley-Tukey algorithm with HPX parallelism */
    void fft(std::vector<std::complex<double>> &x) {
        const size_t n = x.size();
        
        // Bit-reversal permutation (sequential, O(n))
        size_t j = 0;
        for (size_t i = 0; i < n - 1; i++) {
            if (i < j) {
                std::swap(x[i], x[j]);
            }
            size_t k = n >> 1;
            while (k <= j) {
                j -= k;
                k >>= 1;
            }
            j += k;
        }
        
        // Butterfly operations - parallel across independent butterfly groups
        for (size_t len = 2; len <= n; len <<= 1) {
            double angle = -2.0 * M_PI / len;
            std::complex<double> wlen(std::cos(angle), std::sin(angle));
            
            // Launch parallel tasks for independent butterfly groups
            std::vector<hpx::future<void>> futures;
            futures.reserve(n / len);
            
            for (size_t i = 0; i < n; i += len) {
                futures.push_back(hpx::async([&x, i, len, wlen]() {
                    std::complex<double> w(1.0, 0.0);
                    size_t half = len >> 1;
                    for (size_t j = 0; j < half; j++) {
                        std::complex<double> u = x[i + j];
                        std::complex<double> v = x[i + j + half] * w;
                        x[i + j] = u + v;
                        x[i + j + half] = u - v;
                        w *= wlen;
                    }
                }));
            }
            hpx::wait_all(futures);
        }
    }
    ```

    **Solution Overview**
    
    The inverse FFT can be obtained from a forward FFT by the well‑known “conjugate‑FFT‑conjugate‑scale” recipe:
    
    1. Conjugate every element of the input.  
    2. Run the **forward** FFT (the existing `fft` routine).  
    3. Conjugate the result again.  
    4. Scale every element by `1/N` (`N` = number of points).
    
    All three point‑wise stages can be parallelised with HPX.  
    Below is a complete, self‑contained implementation that uses `hpx::for_loop` (the HPX parallel range‑based loop) to perform the conjugation and scaling steps in parallel, while leaving the actual FFT to the supplied `fft` routine.
    
    ```cpp
    // ifft.cpp
    // --------------------------------------------------------------------
    // Compute the inverse Fourier transform in‑place using HPX parallelism.
    // --------------------------------------------------------------------
    
    // HPX headers
    #include <hpx/hpx.hpp>
    #include <hpx/parallel/algorithm.hpp>
    
    // Standard headers
    #include <complex>
    #include <vector>
    
    // --------------------------------------------------------------------
    // Forward FFT – assumed to be implemented elsewhere.
    // It computes an in‑place Cooley‑Tukey FFT (or any other algorithm)
    // on the supplied vector of complex numbers.
    // --------------------------------------------------------------------
    void fft(std::vector<std::complex<double>>& x);
    
    // --------------------------------------------------------------------
    // Inverse FFT – in‑place, parallelised with HPX.
    // --------------------------------------------------------------------
    void ifft(std::vector<std::complex<double>>& x)
    {
        const std::size_t n = x.size();
        if (n == 0) return;                // nothing to do
    
        namespace exec = hpx::execution;
        namespace par  = hpx::parallel;
    
        // ----------------------------------------------------------------
        // 1️⃣  Conjugate the input.
        // ----------------------------------------------------------------
        hpx::for_loop(exec::par, 0, n,
            [&x](std::size_t i)
            {
                x[i] = std::conj(x[i]);
            });
    
        // ----------------------------------------------------------------
        // 2️⃣  Forward FFT (in‑place).  The user‑provided routine is assumed
        //      to be thread‑safe and may already contain its own parallelism.
        // ----------------------------------------------------------------
        fft(x);
    
        // ----------------------------------------------------------------
        // 3️⃣  Conjugate the result (undo the first conjugation).
        // ----------------------------------------------------------------
        hpx::for_loop(exec::par, 0, n,
            [&x](std::size_t i)
            {
                x[i] = std::conj(x[i]);
            });
    
        // ----------------------------------------------------------------
        // 4️⃣  Scale by 1/N to obtain the true inverse transform.
        // ----------------------------------------------------------------
        const double scale = 1.0 / static_cast<double>(n);
        hpx::for_loop(exec::par, 0, n,
            [&x, scale](std::size_t i)
            {
                x[i] *= scale;
            });
    }
    ```

    ### How It Works
    * **Conjugation** – `std::conj` flips the sign of the imaginary part. Doing it before and after the forward FFT yields the mathematical definition of the inverse transform.
    * **Scaling** – The forward FFT does not include the `1/N` factor; dividing by `N` after the second conjugation gives the correct IFFT result.
    * **Parallelism** – Each point‑wise pass (conjugation and scaling) is executed in parallel using `hpx::for_loop` with the `hpx::execution::par` policy. This distributes the loop iterations across all available HPX worker threads.
